/**

export interface TimeSeriesDataPoint {
  /** ISO 8601 date string for the interval */

  /** Aggregated value */

  /** Optional: additional metrics for this point */
  metadata?: {
    count?: number;
    uniqueCount?: number;
    avgValue?: number;
  };
}
