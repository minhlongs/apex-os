/**
 * Get emoji based on severity
 */
function getSeverityEmoji(severity: string): string {
  switch (severity?.toLowerCase()) {
    case 'critical':
      return '🚨';
    case 'high':
      return '⚠️';
    case 'warning':
      return '🔶';
    case 'info':
      return 'ℹ️';
    default:
      return '📢';
  }
}
